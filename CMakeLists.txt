cmake_minimum_required(VERSION 3.20)
project(PracticeEDA VERSION 0.4.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(EDA_BUILD_GUI "Build the wxWidgets editor" ON)
set_property(GLOBAL PROPERTY USE_FOLDERS ON)
set_property(GLOBAL PROPERTY PREDEFINED_TARGETS_FOLDER "CMake")

# List headers explicitly so they appear in generated Visual Studio projects.
set(EDA_CORE_HEADERS include/LogicSupport.h include/Digital.h)
set(EDA_CORE_SOURCES src/LogicSupport.cpp src/Digital.cpp src/CustomComponents.cpp src/LogisimImport.cpp)
set(EDA_GUI_HEADERS include/WxSupport.h include/LogicIcons.h include/LogicSymbols.h include/DigitalEditor.h)
set(EDA_GUI_SOURCES src/App.cpp src/LogicIcons.cpp src/LogicSymbols.cpp src/DigitalEditor.cpp)
set(EDA_DOCUMENTS README.md docs/VISUAL_STUDIO_CN.md docs/DIGITAL_LOGIC_CN.md
    docs/CUSTOM_COMPONENTS_NETLIST_CN.md docs/LOGISIM_IMPORT_CN.md docs/CODE_GUIDE_CN.md docs/TEST_REPORT_CN.md resources/ICONS.md)
set(EDA_EXAMPLES examples/half-adder.logic.json examples/bus-register.logic.json
    examples/hierarchical-adder.logic.json examples/half-adder.component.json examples/logisim-limited.circ)
file(GLOB EDA_ICON_RESOURCES CONFIGURE_DEPENDS "resources/wxwidgets/*" "resources/reference/*")
set_source_files_properties(${EDA_ICON_RESOURCES} PROPERTIES HEADER_FILE_ONLY TRUE)
source_group("图标资源" FILES ${EDA_ICON_RESOURCES})
source_group("头文件" FILES ${EDA_CORE_HEADERS} ${EDA_GUI_HEADERS})
source_group("源文件" FILES ${EDA_CORE_SOURCES} ${EDA_GUI_SOURCES})
source_group("源文件" FILES tests/digital_tests.cpp tests/component_netlist_tests.cpp tests/logisim_import_tests.cpp)
source_group("第三方文件" FILES third_party/json.hpp third_party/tinyxml2.h third_party/tinyxml2.cpp)
source_group("文档" FILES ${EDA_DOCUMENTS})
source_group("示例工程" FILES ${EDA_EXAMPLES})
source_group("构建配置" FILES CMakePresets.json scripts/build.ps1)
if(MSVC)
  add_compile_options(/utf-8 /W4 /permissive-)
else()
  add_compile_options(-Wall -Wextra -Wpedantic)
endif()
add_library(logic_core ${EDA_CORE_HEADERS} ${EDA_CORE_SOURCES} third_party/json.hpp third_party/tinyxml2.h third_party/tinyxml2.cpp)
target_include_directories(logic_core PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/include")
target_include_directories(logic_core SYSTEM PRIVATE third_party)
target_compile_definitions(logic_core PUBLIC NOMINMAX)
set_target_properties(logic_core PROPERTIES FOLDER "库")
if(EDA_BUILD_GUI)
  # Discover the existing sibling dependency for both VS Open Folder and .sln workflows.
  # Explicit -DwxWidgets_ROOT_DIR / cache values always take precedence.
  if(WIN32 AND NOT wxWidgets_ROOT_DIR AND EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/../wxWidgets/include/wx/wx.h")
    get_filename_component(_eda_wx_root "${CMAKE_CURRENT_SOURCE_DIR}/../wxWidgets" ABSOLUTE)
    set(wxWidgets_ROOT_DIR "${_eda_wx_root}" CACHE PATH "wxWidgets installation directory")
  endif()
  if(MSVC AND CMAKE_SIZEOF_VOID_P EQUAL 8 AND NOT wxWidgets_LIB_DIR
      AND EXISTS "${wxWidgets_ROOT_DIR}/lib/vc_x64_lib")
    set(wxWidgets_LIB_DIR "${wxWidgets_ROOT_DIR}/lib/vc_x64_lib" CACHE PATH "wxWidgets library directory")
  endif()
  set(wxWidgets_USE_STATIC ON)
  find_package(wxWidgets 3.2 REQUIRED COMPONENTS core base)
  include(${wxWidgets_USE_FILE})
  add_executable(PracticeEDA WIN32 ${EDA_GUI_HEADERS} ${EDA_CORE_HEADERS} ${EDA_GUI_SOURCES}
      ${EDA_DOCUMENTS} ${EDA_EXAMPLES} ${EDA_ICON_RESOURCES} CMakePresets.json scripts/build.ps1)
  target_include_directories(PracticeEDA PRIVATE resources)
  target_link_libraries(PracticeEDA PRIVATE logic_core ${wxWidgets_LIBRARIES})
  set_target_properties(PracticeEDA PROPERTIES VS_DEBUGGER_WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}")
  set_property(DIRECTORY PROPERTY VS_STARTUP_PROJECT PracticeEDA)
endif()
option(BUILD_TESTING "Build regression tests" ON)
if(BUILD_TESTING)
  enable_testing()
  add_executable(logic_tests tests/digital_tests.cpp)
  target_link_libraries(logic_tests PRIVATE logic_core)
  target_compile_definitions(logic_tests PRIVATE LOGIC_EXAMPLES_DIR="${CMAKE_CURRENT_SOURCE_DIR}/examples")
  set_target_properties(logic_tests PROPERTIES FOLDER "测试")
  add_test(NAME digital_logic_tests COMMAND logic_tests)
  # Keep feature tests separate from the pre-existing simulation regression suite.
  add_executable(component_netlist_tests tests/component_netlist_tests.cpp)
  target_link_libraries(component_netlist_tests PRIVATE logic_core)
  target_include_directories(component_netlist_tests SYSTEM PRIVATE third_party)
  target_compile_definitions(component_netlist_tests PRIVATE LOGIC_EXAMPLES_DIR="${CMAKE_CURRENT_SOURCE_DIR}/examples")
  set_target_properties(component_netlist_tests PROPERTIES FOLDER "测试")
  add_test(NAME component_netlist_tests COMMAND component_netlist_tests "${CMAKE_CURRENT_BINARY_DIR}/qa-features/$<CONFIG>")
  add_executable(logisim_import_tests tests/logisim_import_tests.cpp)
  target_link_libraries(logisim_import_tests PRIVATE logic_core)
  target_compile_definitions(logisim_import_tests PRIVATE LOGIC_EXAMPLES_DIR="${CMAKE_CURRENT_SOURCE_DIR}/examples")
  set_target_properties(logisim_import_tests PROPERTIES FOLDER "测试")
  add_test(NAME logisim_import_tests COMMAND logisim_import_tests)
  find_package(Python3 QUIET COMPONENTS Interpreter)
  if(Python3_Interpreter_FOUND)
    add_test(NAME netlist_format_tests COMMAND "${Python3_EXECUTABLE}"
      "${CMAKE_CURRENT_SOURCE_DIR}/tests/verify_netlists.py" "${CMAKE_CURRENT_BINARY_DIR}/qa-features/$<CONFIG>")
    set_tests_properties(netlist_format_tests PROPERTIES DEPENDS component_netlist_tests)
  endif()
endif()
if(EDA_BUILD_GUI)
  install(TARGETS PracticeEDA RUNTIME DESTINATION .)
  install(FILES ${EDA_EXAMPLES} DESTINATION examples)
  install(FILES README.md DESTINATION .)
  install(FILES scripts/DigitalLogic.cmd DESTINATION .)
  install(FILES docs/VISUAL_STUDIO_CN.md docs/DIGITAL_LOGIC_CN.md docs/CUSTOM_COMPONENTS_NETLIST_CN.md docs/LOGISIM_IMPORT_CN.md
      docs/CODE_GUIDE_CN.md docs/TEST_REPORT_CN.md resources/ICONS.md DESTINATION docs)
  install(FILES resources/ICONS.md DESTINATION resources)
  install(DIRECTORY third_party/ DESTINATION licenses FILES_MATCHING PATTERN "LICENSE*.txt")
endif()
